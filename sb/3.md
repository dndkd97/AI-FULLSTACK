-- ============================================
-- 채용공고
-- ============================================
CREATE TABLE recruit (
    rec_id          NUMBER(10)      NOT NULL,              -- 채용공고 ID
    com_id          NUMBER(10)      NOT NULL,              -- 회사 ID
    title           VARCHAR2(200)   NOT NULL,              -- 채용공고 제목
    department      VARCHAR2(100),                          -- 채용 부서
    position        VARCHAR2(100),                          -- 채용 직무/직급
    headcount       NUMBER,                                 -- 모집 인원
    employment_type VARCHAR2(50),                           -- 고용 형태 (정규직/계약직/인턴 등)
    description     CLOB,                                   -- 채용공고 상세 내용
    start_date      DATE,                                   -- 모집 시작일
    end_date        DATE,                                   -- 모집 마감일
    status          VARCHAR2(20)    DEFAULT 'OPEN' NOT NULL,-- 채용공고 상태
                                                            -- OPEN: 모집중
                                                            -- CLOSED: 마감
                                                            -- CANCELLED: 취소
    created_by      NUMBER(10)      NOT NULL,              -- 채용공고 등록자(사원 ID)
    created_at      DATE            DEFAULT SYSDATE NOT NULL, -- 채용공고 등록일
    updated_at      DATE            DEFAULT SYSDATE NOT NULL, -- 채용공고 수정일
    
    CONSTRAINT pk_recruit PRIMARY KEY (rec_id),
    CONSTRAINT fk_recruit_com FOREIGN KEY (com_id) REFERENCES company(com_id),
    CONSTRAINT fk_recruit_emp FOREIGN KEY (created_by) REFERENCES employee(emp_id),
    CONSTRAINT ck_recruit_status CHECK (status IN ('OPEN', 'CLOSED', 'CANCELLED'))
);

-- 채용공고 PK 자동 생성용 시퀀스
CREATE SEQUENCE seq_recruit START WITH 1 INCREMENT BY 1 NOCACHE NOCYCLE;

-- 채용공고 등록 시 PK 자동 생성
CREATE OR REPLACE TRIGGER trg_recruit_pk
BEFORE INSERT ON recruit
FOR EACH ROW
BEGIN
    IF :NEW.rec_id IS NULL THEN
        :NEW.rec_id := seq_recruit.NEXTVAL;
    END IF;
END;
/

-- 채용공고 수정 시 수정일 자동 갱신
CREATE OR REPLACE TRIGGER trg_recruit_bu
BEFORE UPDATE ON recruit
FOR EACH ROW
BEGIN
    :NEW.updated_at := SYSDATE;
END;
/

-- ============================================
-- 지원자
-- ============================================
CREATE TABLE applicant (
    applicant_id    NUMBER(10)      NOT NULL,              -- 지원자 ID
    com_id          NUMBER(10)      NOT NULL,              -- 회사 ID
    rec_id          NUMBER(10)      NOT NULL,              -- 지원한 채용공고 ID
    name            VARCHAR2(50)    NOT NULL,              -- 지원자 이름
    email           VARCHAR2(100),                          -- 지원자 이메일
    phone           VARCHAR2(20),                           -- 지원자 연락처
    resume_url      VARCHAR2(500),                          -- 이력서 파일 경로/URL
    status          VARCHAR2(20)    DEFAULT 'RECEIVED' NOT NULL,
                                                            -- 지원 상태
                                                            -- RECEIVED: 지원 접수
                                                            -- SCREENING: 서류 검토
                                                            -- INTERVIEW: 면접
                                                            -- HIRED: 최종 합격
                                                            -- REJECTED: 불합격

    applied_date    DATE            DEFAULT SYSDATE,        -- 지원일
    created_at      DATE            DEFAULT SYSDATE NOT NULL, -- 지원자 등록일
    updated_at      DATE            DEFAULT SYSDATE NOT NULL, -- 지원자 수정일

    CONSTRAINT pk_applicant PRIMARY KEY (applicant_id),
    CONSTRAINT fk_applicant_com FOREIGN KEY (com_id) REFERENCES company(com_id),
    CONSTRAINT fk_applicant_recruit FOREIGN KEY (rec_id) REFERENCES recruit(rec_id),
    CONSTRAINT ck_applicant_status CHECK ( status IN ( 'RECEIVED', 'SCREENING', 'INTERVIEW', 'HIRED', 'REJECTED' ) )
);

BEFORE INSERT ON applicant
FOR EACH ROW
BEGIN
    IF :NEW.applicant_id IS NULL THEN
        :NEW.applicant_id := seq_applicant.NEXTVAL;
    END IF;
END;
/

-- 지원자 수정 시 수정일 자동 갱신
CREATE OR REPLACE TRIGGER trg_applicant_bu
BEFORE UPDATE ON applicant
FOR EACH ROW
BEGIN
    :NEW.updated_at := SYSDATE;
END;
/

-- 채용공고별 지원자 조회 성능 향상을 위한 인덱스
CREATE INDEX fk_applicant_recruit_idx ON applicant(rec_id);

-- 지원 상태별 조회 성능 향상을 위한 인덱스
CREATE INDEX idx_applicant_status ON applicant(status);

-- 회사별 채용공고 조회 성능 향상을 위한 인덱스
CREATE INDEX idx_recruit_com_id ON recruit(com_id);




-- ============================================
-- 이력서
-- ============================================

CREATE TABLE resume (
    resume_id       NUMBER(10)      NOT NULL,              -- 이력서 ID
    applicant_id    NUMBER(10)      NOT NULL,              -- 지원자 ID
    file_name       VARCHAR2(200),                          -- 이력서 파일명
    file_url        VARCHAR2(500),                          -- 이력서 파일 경로/URL
    extracted_text  CLOB,                                   -- 이력서에서 추출한 전체 텍스트
    ai_summary      CLOB,                                   -- AI가 생성한 이력서 요약
    fit_score       NUMBER,                                 -- 채용공고와 지원자의 적합도 점수
    status          VARCHAR2(20)    DEFAULT 'PENDING' NOT NULL,
                                                            -- AI 분석 상태
                                                            -- PENDING: 분석 대기
                                                            -- COMPLETED: 분석 완료
                                                            -- FAILED: 분석 실패

    uploaded_at     DATE            DEFAULT SYSDATE NOT NULL, -- 이력서 업로드일
    analyzed_at     DATE,                                   -- AI 분석 완료일

    CONSTRAINT pk_resume PRIMARY KEY (resume_id),
    CONSTRAINT fk_resume_applicant FOREIGN KEY (applicant_id) REFERENCES applicant(applicant_id),
    CONSTRAINT ck_resume_status CHECK ( status IN ( 'PENDING', 'COMPLETED', 'FAILED' ) )
);


-- 이력서 PK 자동 생성용 시퀀스
CREATE SEQUENCE seq_resume START WITH 1 INCREMENT BY 1 NOCACHE NOCYCLE;

-- 이력서 등록 시 PK 자동 생성
CREATE OR REPLACE TRIGGER trg_resume_pk
BEFORE INSERT ON resume
FOR EACH ROW
BEGIN
    IF :NEW.resume_id IS NULL THEN
        :NEW.resume_id := seq_resume.NEXTVAL;
    END IF;
END;
/



-- ============================================
-- 이력서 청크 + 임베딩 (RAG)
-- ============================================

CREATE TABLE resume_chunk (
    chunk_id        NUMBER(10)      NOT NULL,              -- 청크 ID
    resume_id       NUMBER(10)      NOT NULL,              -- 이력서 ID
    chunk_order     NUMBER          NOT NULL,              -- 이력서 내 청크 순서
    chunk_text      CLOB            NOT NULL,              -- 분할된 이력서 텍스트
    embedding_json  CLOB            NOT NULL,              -- 텍스트를 벡터화한 임베딩 데이터(JSON)
    CONSTRAINT pk_resume_chunk PRIMARY KEY (chunk_id),
    CONSTRAINT fk_chunk_resume FOREIGN KEY (resume_id) REFERENCES resume(resume_id)
);


-- 이력서 청크 PK 자동 생성용 시퀀스
CREATE SEQUENCE seq_resume_chunk START WITH 1 INCREMENT BY 1 NOCACHE NOCYCLE;

-- 이력서별 청크 조회 성능 향상을 위한 인덱스
CREATE INDEX fk_chunk_resume_idx ON resume_chunk(resume_id);
